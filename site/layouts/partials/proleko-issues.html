{{/* Every Proleko? issue, newest first. data/proleko/ holds one file per issue, named year-number. Each file is year: 2026 number: 3 title: "Proleko? 60" # empty when the issue has no name pdf: 2026/proleko-2026-3.pdf cover: images/proleko/kannet/2026-3.webp Files are written by tools/import-proleko.py for the old archive and by Decap (the proleko collection) for every new issue. The PDFs are not in the repository: at 1.3 GB, some over GitHub's file limit, they are served from the server's disk under params.prolekoMedia. An issue's cover is either an image in assets (`cover`, which Hugo resizes) or, when that is empty, the JPEG the proxy stored next to the PDF when it was uploaded through Decap. Returns a slice of maps with the file's fields plus slug, url (the PDF), coverURL (the stored JPEG, for an issue without `cover`), label ("3/2026") and name (the title, or the label for an untitled one). */}} {{ $media := site.Params.prolekoMedia | default "/media/proleko/" }} {{ $out := slice }} {{ range $slug, $i := hugo.Data.proleko }} {{ $label := printf "%d/%d" (int $i.number) (int $i.year) }} {{ $url := printf "%s%s" $media $i.pdf }} {{/* TEMPORARY until Prodeko/infrastructure#43: an old issue links its PDF where the old site kept it, because /media/proleko/ is not served yet. An issue uploaded through Decap is not in the map and keeps $url. */}} {{ with site.Params.prolekoStatic }}{{ with index hugo.Data.prolekostatic $slug }} {{ $url = printf "%s%s" site.Params.prolekoStatic . }} {{ end }}{{ end }} {{ $out = $out | append (merge $i (dict "slug" $slug "sortKey" (add (mul (int $i.year) 100) (int $i.number)) "url" $url "coverURL" (printf "%s%s.jpg" $media (strings.TrimSuffix ".pdf" $i.pdf)) "label" $label "name" (or $i.title $label))) }} {{ end }} {{ return sort $out "sortKey" "desc" }}
